Postgres

Postgres Compose

Achtung: Passwort eintragen.

networks:
  adbkt:
    external: true
   
services:
  pg:
    container_name: pg
    image: postgres:latest
    restart: unless-stopped
    # shm_size: 1g
    environment:
      POSTGRES_PASSWORD: #<-- hier Passwort eintragen
      PGDATA: /var/lib/postgresql/data/pgdata
    volumes:
      - data:/var/lib/postgresql/data

    networks:
      - adbkt
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres -d postgres"]
      interval: 5s
      timeout: 5s
      retries: 3
      
volumes:
  data:

Python

Per Python auf Postgres zugreifen

  • Python-File-Verzeichnis: /home/du/adbkt1/python
  • Python-File: pg-test.py
  • Umgebung aktivieren: source /home/du/.py/bin/activate
  • in Python-Verzeichnis wechseln
  • Python-File ausführen: python pg-test.py
import psycopg

conninfo = " ".join([
"user='postgres'",
"password=''",   #<-- hier Passwort eintragen
"host='pg'",
"port=5432",
"dbname='postgres'"])

sql1 = "drop table if exists kv"
sql2 = """
create table kv (
  k integer not null,
  v varchar not null
)
"""
sql3 = "insert into kv values (1, 'hallo'), (2, 'hi')"

with psycopg.connect(conninfo) as conn:
    conn.execute(sql1)
    conn.execute(sql2)
    conn.execute(sql3)

sql = "select * from kv"
with psycopg.connect(conninfo) as conn:
    rs = conn.execute(sql).fetchall()

print(rs)
